ESP32 & MQTT Architecture Guide

Complete technical reference for connecting your physical hardware to TerraWatch

System Pipeline

[1. Sensors] MPU6050 (Tilt) + SW-420 (Vib) + Soil Moisture + Rain Gauge
↓ Analog/I2C Signals
[2. ESP32 Node] Sampling, Kalman filtering, JSON serialization
↓ MQTT Publish over TCP (1883) or TLS (8883)
[3. MQTT Broker] Mosquitto / HiveMQ / EMQX with WebSocket listener enabled
↓ MQTT over WebSockets (ws:// / wss://)
[4. TerraWatch UI] Real-time subscriber, geotechnical risk engine & alert siren

ESP32 Arduino C++ Code

PubSubClient + ArduinoJson 
#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <Wire.h>

const char* ssid = "YOUR_WIFI";
const char* password = "YOUR_PASSWORD";
const char* mqtt_server = "broker.hivemq.com";
const int mqtt_port = 1883;
const char* topic = "landslide/node01/telemetry";

WiFiClient espClient;
PubSubClient client(espClient);

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();

  // Read Sensors
  float tilt = readMPU6050Tilt();
  float vibration = readVibration();
  float moisture = readSoilMoisture();
  float rain = readRainfall();

  // Serialize to JSON
  StaticJsonDocument<200> doc;
  doc["tilt"] = tilt;
  doc["vibration"] = vibration;
  doc["soil_moisture"] = moisture;
  doc["rain"] = rain;
  doc["node_id"] = "ESP32_NODE_01";

  char buffer[256];
  serializeJson(doc, buffer);
  client.publish(topic, buffer);

  delay(2000); // 2 second telemetry interval
}
